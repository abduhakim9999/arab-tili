يَسَرَ يَسَرَا يَسَرُوا يَسَرَتْ يَسَرَتَا يَسَرْنَ يَسَرْتَ يَسَرْتُمَا يَسَرْتُمْ يَسَرْتِ يَسَرْتُمَا يَسَرْتُنَّ يَسَرْتُ يَسَرْنَا يُسِرَ يُسِرَا يُسِرُوا يُسِرَتْ يُسِرَتَا يُسِرْنَ يُسِرْتَ يُسِرْتُمَا يُسِرْتُمْ يُسِرْتِ يُسِرْتُمَا يُسِرْتُنَّ يُسِرْتُ يُسِرْنَا يَيْسِرُ يَيْسِرَانِ يَيْسِرُونَ تَيْسِرُ تَيْسِرَانِ يَيْسِرْنَ تَيْسِرُ تَيْسِرَانِ تَيْسِرُونَ تَيْسِرِينَ تَيْسِرَانِ تَيْسِرْنَ أَيْسِرُ نَيْسِرُ يُوسَرُ يُوسَرَانِ يُوسَرُونَ تُوسَرُ تُوسَرَانِ يُوسَرْنَ تُوسَرُ تُوسَرَانِ تُوسَرُونَ تُوسَرِينَ تُوسَرَانِ تُوسَرْنَ أُوسَرُ نُوسَرُ يَاسِرٌ يَاسِرَانِ يَاسِرُونَ يَاسِرَةٌ يَاسِرَتَانِ يَاسِرَاتٌ مَيْسُورٌ مَيْسُورَانِ مَيْسُورُونَ مَيْسُورَةٌ مَيْسُورَتَانِ مَيْسُورَاتٌ لَمْ يَيْسِرْ لَمْ يَيْسِرَا لَمْ يَيْسِرُوا لَمْ تَيْسِرْ لَمْ تَيْسِرَا لَمْ يَيْسِرْنَ لَمْ تَيْسِرْ لَمْ تَيْسِرَا لَمْ تَيْسِرُوا لَمْ تَيْسِرِي لَمْ تَيْسِرَا لَمْ تَيْسِرْنَ لَمْ أَيْسِرْ لَمْ نَيْسِرْ لَا يَيْسِرُ لَا يَيْسِرَانِ لَا يَيْسِرُونَ لَا تَيْسِرُ لَا تَيْسِرَانِ لَا يَيْسِرْنَ لَا تَيْسِرُ لَا تَيْسِرَانِ لَا تَيْسِرُونَ لَا تَيْسِرِينَ لَا تَيْسِرَانِ لَا تَيْسِرْنَ لَا أَيْسِرُ لَا نَيْسِرُ اِيسِرْ اِيسِرَا اِيسِرُوا اِيسِرِي اِيسِرَا اِيسِرْنَ لِيَيْسِرْ لِيَيْسِرَا لِيَيْسِرُوا لِتَيْسِرْ لِتَيْسِرَا لِيَيْسِرْنَ لِأَيْسِرْ لِنَيْسِرْ لَا يَيْسِرْ لَا يَيْسِرَا لَا يَيْسِرُوا لَا تَيْسِرْ لَا تَيْسِرَا لَا يَيْسِرْنَ لَا تَيْسِرْ لَا تَيْسِرَا لَا تَيْسِرُوا لَا تَيْسِرِي لَا تَيْسِرَا لَا تَيْسِرْنَ لَا أَيْسِرْ لَا نَيْسِرْ لِيُوسَرْ لِيُوسَرَا لِيُوسَرُوا لِتُوسَرْ لِتُوسَرَا لِيُوسَرْنَ لِتُوسَرْ لِتُوسَرَا لِتُوسَرُوا لِتُوسَرِي لِتُوسَرَا لِتُوسَرْنَ لِأُوسَرْ لِنُوسَرْ لَا يُوسَرْ لَا يُوسَرَا لَا يُوسَرُوا لَا تُوسَرْ لَا تُوسَرَا لَا يُوسَرْنَ لَا تُوسَرْ لَا تُوسَرَا لَا تُوسَرُوا لَا تُوسَرِي لَا تُوسَرَا لَا تُوسَرْنَ لَا أُوسَرْ لَا نُوسَرْ مَيْسِرٌ مَيْسِرَانِ مَيَاسِرُ مِيسَارٌ مِيسَارَانِ مَيَاسِيرُ أَيْسَرُ أَيْسَرَانِ أَيْسَرُونَ يُسْرَى يُسْرَيَانِ يُسْرَيَاتٌ
