دَامَ دَامَا دَامُوا دَامَتْ دَامَتَا دُمْنَ دُمْتَ دُمْتُمَا دُمْتُمْ دُمْتِ دُمْتُمَا دُمْتُنَّ دُمْتُ دُمْنَا دِيمَ دِيمَا دِيمُوا دِيمَتْ دِيمَتَا دِمْنَ دِمْتَ دِمْتُمَا دِمْتُمْ دِمْتِ دِمْتُمَا دِمْتُنَّ دِمْتُ دِمْنَا يَدُومُ يَدُومَانِ يَدُومُونَ تَدُومُ تَدُومَانِ يَدُمْنَ تَدُومُ تَدُومَانِ تَدُومُونَ تَدُومِينَ تَدُومَانِ تَدُمْنَ أَدُومُ نَدُومُ يُدَامُ يُدَامَانِ يُدَامُونَ تُدَامُ تُدَامَانِ يُدَمْنَ تُدَامُ تُدَامَانِ تُدَامُونَ تُدَامِينَ تُدَامَانِ تُدَمْنَ أُدَامُ نُدَامُ دَائِمٌ دَائِمَانِ دَائِمُونَ دَائِمَةٌ دَائِمَتَانِ دَائِمَاتٌ مَدُومٌ مَدُومَانِ مَدُومُونَ مَدُومَةٌ مَدُومَتَانِ مَدُومَاتٌ لَمْ يَدُمْ لَمْ يَدُومَا لَمْ يَدُومُوا لَمْ تَدُمْ لَمْ تَدُومَا لَمْ يَدُمْنَ لَمْ تَدُمْ لَمْ تَدُومَا لَمْ تَدُومُوا لَمْ تَدُومِي لَمْ تَدُومَا لَمْ تَدُمْنَ لَمْ أَدُمْ لَمْ نَدُمْ لَا يَدُومُ لَا يَدُومَانِ لَا يَدُومُونَ لَا تَدُومُ لَا تَدُومَانِ لَا يَدُمْنَ لَا تَدُومُ لَا تَدُومَانِ لَا تَدُومُونَ لَا تَدُومِينَ لَا تَدُومَانِ لَا تَدُمْنَ لَا أَدُومُ لَا نَدُومُ دُمْ دُومَا دُومُوا دُومِي دُومَا دُمْنَ لِيَدُمْ لِيَدُومَا لِيَدُومُوا لِتَدُمْ لِتَدُومَا لِيَدُمْنَ لِأَدُمْ لِنَدُمْ لَا يَدُمْ لَا يَدُومَا لَا يَدُومُوا لَا تَدُمْ لَا تَدُومَا لَا يَدُمْنَ لَا تَدُمْ لَا تَدُومَا لَا تَدُومُوا لَا تَدُومِي لَا تَدُومَا لَا تَدُمْنَ لَا أَدُمْ لَا نَدُمْ لِيُدَمْ لِيُدَامَا لِيُدَامُوا لِتُدَمْ لِتُدَامَا لِيُدَمْنَ لِتُدَمْ لِتُدَامَا لِتُدَامُوا لِتُدَامِي لِتُدَامَا لِتُدَمْنَ لِأُدَمْ لِنُدَمْ لَا يُدَمْ لَا يُدَامَا لَا يُدَامُوا لَا تُدَمْ لَا تُدَامَا لَا يُدَمْنَ لَا تُدَمْ لَا تُدَامَا لَا تُدَامُوا لَا تُدَامِي لَا تُدَامَا لَا تُدَمْنَ لَا أُدَمْ لَا نُدَمْ مَدَامٌ مَدَامَانِ مَدَائِمُ مِدَامٌ مِدَامَانِ مَدَائِمُ أَدْوَمُ أَدْوَمَانِ أَدْوَمُونَ دُومَى دُومَيَانِ دُومَيَاتٌ
