ar-SA-HamedNeural
نَصَرَ يَنْصُرُ كَتَبَ يَكْتُبُ دَخَلَ يَدْخُلُ خَرَجَ يَخْرُجُ طَلَبَ يَطْلُبُ تَرَكَ يَتْرُكُ حَضَرَ يَحْضُرُ سَكَنَ يَسْكُنُ دَرَسَ يَدْرُسُ شَكَرَ يَشْكُرُ ذَكَرَ يَذْكُرُ قَتَلَ يَقْتُلُ نَظَرَ يَنْظُرُ سَجَدَ يَسْجُدُ عَبَدَ يَعْبُدُ رَزَقَ يَرْزُقُ حَكَمَ يَحْكُمُ قَعَدَ يَقْعُدُ بَلَغَ يَبْلُغُ خَلَقَ يَخْلُقُ صَدَقَ يَصْدُقُ سَكَتَ يَسْكُتُ طَبَخَ يَطْبُخُ كَنَسَ يَكْنُسُ أَخَذَ يَأْخُذُ أَكَلَ يَأْكُلُ أَمَرَ يَأْمُرُ رَدَّ يَرُدُّ مَدَّ يَمُدُّ شَدَّ يَشُدُّ عَدَّ يَعُدُّ مَرَّ يَمُرُّ دَلَّ يَدُلُّ حَجَّ يَحُجُّ ظَنَّ يَظُنُّ ضَمَّ يَضُمُّ شَكَّ يَشُكُّ وَصَّ يَوُصُّ دَامَ يَدُومُ قَالَ يَقُولُ كَانَ يَكُونُ قَامَ يَقُومُ زَارَ يَزُورُ صَامَ يَصُومُ طَافَ يَطُوفُ مَاتَ يَمُوتُ عَادَ يَعُودُ قَادَ يَقُودُ فَازَ يَفُوزُ تَابَ يَتُوبُ ذَاقَ يَذُوقُ دَعَا يَدْعُو رَجَا يَرْجُو تَلَا يَتْلُو نَجَا يَنْجُو شَكَا يَشْكُو عَفَا يَعْفُو دَنَا يَدْنُو عَلَا يَعْلُو غَزَا يَغْزُو مَحَا يَمْحُو بَدَا يَبْدُو نَمَا يَنْمُو
